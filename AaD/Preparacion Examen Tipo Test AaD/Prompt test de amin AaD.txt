  p.parse(new File("library.xml"), new H());

static class H extends DefaultHandler {
  boolean inTitle=false, inPrice=false, bookPrimary=false, priceEUR=false;
  StringBuilder title = new StringBuilder(), price = new StringBuilder();
  int sum=0;
  public void startElement(String u,String l,String q,Attributes a){
    if("book".equals(q)){ title.setLength(0); price.setLength(0); bookPrimary=false; priceEUR=false; }
    if("title".equals(q)){ inTitle=true; }
    if("author".equals(q)){ if("primary".equals(a.getValue("role"))) bookPrimary=true; }
    if("price".equals(q)){ inPrice=true; priceEUR="EUR".equals(a.getValue("currency")); price.setLength(0); }
  }
  public void characters(char[] ch,int s,int n){
    if(inTitle) title.append(ch,s,n);
    if(inPrice) price.append(ch,s,n);
  }
  public void endElement(String u,String l,String q){
    if("title".equals(q)) inTitle=false;
    if("price".equals(q)) inPrice=false;
    if("book".equals(q)){
      String t = title.toString().trim();
      boolean startsC = !t.isEmpty() && Character.toUpperCase(t.charAt(0))=='C';
      if(startsC && bookPrimary && priceEUR) sum += Integer.parseInt(price.toString().trim());

    if("library".equals(q)) System.out.println(sum);
